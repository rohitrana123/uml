#P2 - hierarchiacal clustering

import numpy as np
import matplotlib.pyplot as plt
from scipy.cluster.hierarchy import dendrogram, linkage
from sklearn.cluster import AgglomerativeClustering

data = np.array([7, 10, 20, 28, 35]).reshape(-1, 1)
#labels = ['P1', 'P2', 'P3', 'P4', 'P5']
labels = [f"P{i+1}" for i in range (len(data))]


x = data.flatten() #Converts [[7],[10],...] → [7,10,...]
                   #Scatter plots need 1D values

for method in ['single', 'complete', 'average']:

    # ---- DENDROGRAM ----
    Z = linkage(data, method=method)
    dendrogram(Z, labels=labels)
    plt.title(f"{method.capitalize()} Linkage")
    plt.xlabel('Participants')
    plt.ylabel('Distance')
    plt.show()

    # ---- CLUSTERING ----
    model = AgglomerativeClustering(n_clusters=2, linkage=method)
    clusters = model.fit_predict(data)
    print(method, clusters)

    # ---- PLOTTING CLUSTERS (ADDED PART) ----
    plt.figure()

    plt.scatter(x[clusters == 0], [0]*sum(clusters == 0), label='Cluster 1', s=100)
    plt.scatter(x[clusters == 1], [0]*sum(clusters == 1), label='Cluster 2', s=100)

    # Point labels
    for i in range(len(x)):
        plt.text(x[i] + 0.5 , 0, labels[i])

    plt.title(f"Clusters ({method.capitalize()} Linkage)")
    plt.xlabel("Age")
    plt.yticks([]) #Hides all numerical ticks and labels on the Y-axis. 
    #This is common for 1D plots where the vertical position ((y=0)) carries no information and would otherwise clutter the visual.
    plt.legend()
    plt.ylim(-0.05, 0.05) #Manually sets the Y-axis limits. 
    plt.show()

import numpy as np
import scipy.cluster.hierarchy as sch

# Age data
data = np.array([7, 10, 20, 28, 35]).reshape(-1, 1)

# Participant labels
labels = ['P1', 'P2', 'P3', 'P4', 'P5']

# Simple interpretation function
def interpret_linkage_simple(linkage_matrix, labels):
    clusters = labels.copy()

    for step, row in enumerate(linkage_matrix, start=1):
        c1 = int(row[0])
        c2 = int(row[1])
        dist = row[2]

        print(
            f"Step {step}: Merge {clusters[c1]} and {clusters[c2]} "
            f"at distance {dist:.2f}"
        )

        # Add merged cluster
        clusters.append(f"({clusters[c1]} + {clusters[c2]})")

# Run for all linkage methods
for method in ['single', 'complete', 'average']:
    print(f"\n========== {method.capitalize()} Linkage ==========")
    Z = sch.linkage(data, method=method)
    interpret_linkage_simple(Z, labels)
