import numpy as np
import matplotlib.pyplot as plt
from scipy.cluster.hierarchy import dendrogram, linkage
from sklearn.cluster import AgglomerativeClustering

data = np.array([7, 10, 20, 28, 35]).reshape(-1, 1)
#labels = ['P1', 'P2', 'P3', 'P4', 'P5']
labels = [f"P{i+1}" for i in range (len(data))]


x = data.flatten() #Converts [[7],[10],...] → [7,10,...]
                   #Scatter plots need 1D values

for method in ['single', 'complete', 'average']:

    # ---- DENDROGRAM ----
    Z = linkage(data, method=method)
    dendrogram(Z, labels=labels)
    plt.title(f"{method.capitalize()} Linkage")
    plt.xlabel('Participants')
    plt.ylabel('Distance')
    plt.show()

    # ---- CLUSTERING ----
    model = AgglomerativeClustering(n_clusters=2, linkage=method)
    clusters = model.fit_predict(data)
    print(method, clusters)

    # ---- PLOTTING CLUSTERS (ADDED PART) ----
    plt.figure()

    plt.scatter(x[clusters == 0], [0]*sum(clusters == 0), label='Cluster 1', s=100)
    plt.scatter(x[clusters == 1], [0]*sum(clusters == 1), label='Cluster 2', s=100)

    # Point labels
    for i in range(len(x)):
        plt.text(x[i] + 0.5 , 0, labels[i])

    plt.title(f"Clusters ({method.capitalize()} Linkage)")
    plt.xlabel("Age")
    plt.yticks([]) #Hides all numerical ticks and labels on the Y-axis. 
    #This is common for 1D plots where the vertical position ((y=0)) carries no information and would otherwise clutter the visual.
    plt.legend()
    plt.ylim(-0.05, 0.05) #Manually sets the Y-axis limits. 
    plt.show()

import numpy as np
import scipy.cluster.hierarchy as sch

# Age data
data = np.array([7, 10, 20, 28, 35]).reshape(-1, 1)

# Participant labels
labels = ['P1', 'P2', 'P3', 'P4', 'P5']

# Simple interpretation function
def interpret_linkage_simple(linkage_matrix, labels):
    clusters = labels.copy()

    for step, row in enumerate(linkage_matrix, start=1):
        c1 = int(row[0])
        c2 = int(row[1])
        dist = row[2]

        print(
            f"Step {step}: Merge {clusters[c1]} and {clusters[c2]} "
            f"at distance {dist:.2f}"
        )

        # Add merged cluster
        clusters.append(f"({clusters[c1]} + {clusters[c2]})")

# Run for all linkage methods
for method in ['single', 'complete', 'average']:
    print(f"\n========== {method.capitalize()} Linkage ==========")
    Z = sch.linkage(data, method=method)
    interpret_linkage_simple(Z, labels)

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.cluster import DBSCAN
from sklearn.preprocessing import StandardScaler

# ---------------- DATA ----------------
data = np.array([
    [15, 2], [16, 3], [15.5, 2.5], [18, 4],
    [50, 40], [51, 42], [49, 39],
    [80, 100], [82, 98], [81, 101], [85, 90]
])

df = pd.DataFrame(data, columns=["Annual Spending", "Shopping Frequency"])

# ---------------- SCALING ----------------
scaled_data = StandardScaler().fit_transform(data)

# ---------------- DBSCAN ----------------
dbscan = DBSCAN(eps=0.5, min_samples=3)
df["Cluster"] = dbscan.fit_predict(scaled_data) #This single line executes the algorithm on your scaled_data and stores the results directly into a new column in your DataFrame. 

# ---------------- CORE / NOISE ----------------
df["Type"] = "Border"  #Initializes a new column called "Type" and sets every row to "Border" as a starting default.
df.loc[dbscan.core_sample_indices_, "Type"] = "Core"  # down in img
df.loc[df["Cluster"] == -1, "Type"] = "Noise"   # Finds all rows where the cluster label is -1 and changes their type to "Noise".

print("DBSCAN Results:")
print(df)

# ---------------- PLOTTING ----------------
plt.figure(figsize=(9, 7))

for label in sorted(df["Cluster"].unique()):   #This starts a loop that runs once for every unique cluster ID in your "Cluster" column
    if label == -1:                         #This condition checks if the current iteration is dealing with the special Noise label (-1) from DBSCAN.
        subset = df[df["Cluster"] == -1]   #It filters the DataFrame to get only the noise points.
        plt.scatter(                       #It plots these noise points
            subset["Annual Spending"],
            subset["Shopping Frequency"],
            color="red",
            label="Noise",
            s=100,
            edgecolor="black"
        )
    else:                                    #This executes for all other cluster labels (0, 1, 2, etc.).
        subset = df[df["Cluster"] == label]  #It filters the DataFrame for that specific cluster's points.
        plt.scatter(
            subset["Annual Spending"],
            subset["Shopping Frequency"],
            label=f"Cluster {label}",     #It creates a dynamic label for the legend (e.g., "Cluster 0", "Cluster 1").
            s=100,
            edgecolor="black"
        )

# ---------------- ANNOTATE CORE / BORDER / NOISE ----------------adds text labels
for i, row in df.iterrows():  #This initiates a loop that goes through your DataFrame row by row. In each loop iteration, i holds the row index, and row holds all the data for that specific data point.
    plt.text(
        row["Annual Spending"] + 0.5,
        row["Shopping Frequency"],
        row["Type"],    #This is the actual text content that is displayed for that specific point—either "Core", "Border", or "Noise".
        fontsize=8
    )

plt.title("DBSCAN Clustering with Core, Border, and Noise Points")
plt.xlabel("Annual Spending ($1000s)")
plt.ylabel("Shopping Frequency (times/year)")
plt.legend()
plt.grid(True)
plt.show()

import numpy as np
import matplotlib.pyplot as plt
from sklearn.cluster import KMeans
from sklearn.mixture import GaussianMixture
from sklearn.datasets import make_blobs
from scipy.stats import norm

#🔹 2. Generate synthetic dataset
X, _ = make_blobs(
    n_samples=500,
    centers=3,
    cluster_std=[1.0, 2.5, 0.5],
    random_state=42
)

#🔹 3. Apply K-Means clustering
kmeans = KMeans(n_clusters=3, random_state=42)
k_labels = kmeans.fit_predict(X)

#🔹 4. Apply Gaussian Mixture Model (GMM)
gmm = GaussianMixture(n_components=3, random_state=42)
gmm_labels = gmm.fit_predict(X)

plt.figure(figsize=(12, 5))

# K-Means plot
plt.subplot(1, 2, 1)     #This divides the figure into a grid with 1 row and 2 columns. The final 1 indicates this is the first plot (left side). Matplotlib subplots are indexed starting from 1.
plt.scatter(X[:, 0], X[:, 1], c=k_labels, s=30)    #These are the coordinates. X[:, 0] pulls all values from the first column (x-axis), and X[:, 1] pulls the second column (y-axis).
plt.scatter(kmeans.cluster_centers_[:, 0],         #This attribute stores the final coordinates (x,y) of the 3 centroids found by the Scikit-learn KMeans algorithm.
            kmeans.cluster_centers_[:, 1],
            c='red', marker='X', s=200)
plt.title("K-Means Clustering")

# GMM plot
plt.subplot(1, 2, 2)
plt.scatter(X[:, 0], X[:, 1], c=gmm_labels, s=30)
plt.scatter(gmm.means_[:, 0],
            gmm.means_[:, 1],
            c='red', marker='X', s=200)
plt.title("GMM Clustering")

plt.show()

# 🔹 6. Plot Gaussian curves (ONLY for Feature 1)
x_vals = np.linspace(X[:, 0].min(), X[:, 0].max(), 200)

plt.figure(figsize=(8, 5))

for i in range(3):
    mean = gmm.means_[i, 0]
    std = np.sqrt(gmm.covariances_[i][0, 0])
    plt.plot(x_vals, norm.pdf(x_vals, mean, std),
             label=f"Gaussian {i+1}")

plt.title("Gaussian Curves (Feature 1)")
plt.xlabel("Feature 1")
plt.ylabel("Density")
plt.legend()
plt.show()
