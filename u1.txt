## P1 -Kmeans clustering

-> #Step 1: Load and explore the dataset.
import pandas as pd

#Load the dataset
df = pd.read_csv('Mall_Customers.csv') 
print(df.head())


-> #Step 2: Data preprocessing (selecting relevant features)
# Select 'Annual Income' and 'Spending Score' for clustering
X = df[['Annual Income (k$)' , 'Spending Score (1-100)']]


-> # Step 3: Perform K-Mean Clustering
from sklearn.cluster import KMeans
import matplotlib.pyplot as plt

#Apply K-Means with K=3 (initial guess)
kmeans = KMeans(n_clusters=3, init='k-means++', random_state=42)
y_kmeans = kmeans.fit_predict(X)

#Visualize the result
plt.scatter(X.iloc[:,0], X.iloc[:,1], c=y_kmeans, cmap='viridis')
plt.scatter(kmeans.cluster_centers_[:,0], kmeans.cluster_centers_[:,1], s=200, c='red', marker='x')
plt.xlabel('Annual Income (k$)')
plt.ylabel('Spending Score (1-100)')
plt.title('Customer Segmentation using K-Means')
plt.show()


-> #Step 4: Compare random initialization and K-Means++ initialization

#Using random intialization
kmeans_random = KMeans(n_clusters=3, init='random', random_state=42)
y_kmeans_random = kmeans_random.fit_predict(X)

#compare cluster centers
print('Cluster Centers (K-Means++):' , kmeans.cluster_centers_)
print('Cluster Centers (Random):' , kmeans_random.cluster_centers_)


-> # Choosing the Optimal Number of Clusters
# Elbow Method to determine the optimal number of clusters

inertia = []
for k in range(1, 11):
    kmeans = KMeans(n_clusters=k, init='k-means++', random_state=42)
    kmeans.fit(X)
    inertia.append(kmeans.inertia_)

#Visualize the result
plt.plot(range(1, 11), inertia, marker='o')
plt.xlabel('Numer of clusters (k)')
plt.ylabel('Inertia')
plt.title('Elbow Method')
plt.show()


-> #Evaluation Metrics for Clustering 
#Explain Silhoutee Score for evaluating clustering quality
#Compute the silhouette score for different values of k.

from sklearn.metrics import silhouette_score
for k in range(2, 10):
    kmeans = KMeans (n_clusters=k, init='k-means++', random_state=42) 
    y_kmeans = kmeans.fit_predict(X)
    score = silhouette_score(X,y_kmeans)
    print(f"Silhouette Score for k={k}:{score}")
